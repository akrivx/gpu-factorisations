cmake_minimum_required(VERSION 3.26)
include(cmake/Dependencies.cmake)

# native is convenient for experiments on the build machine. Override for deployment.
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES native CACHE STRING "CUDA architectures")
endif()
project(gpu_factorisations VERSION 0.1.0 LANGUAGES CXX CUDA)
include(cmake/Formatting.cmake)

find_package(CUDAToolkit REQUIRED)
find_package(lapacke CONFIG REQUIRED)

add_library(lu_reference src/lu_reference.cpp)
target_include_directories(lu_reference PUBLIC include)
target_compile_features(lu_reference PUBLIC cxx_std_17)
target_link_libraries(lu_reference PRIVATE lapacke)

add_library(lu_host src/lu_host.cpp)
target_include_directories(lu_host PUBLIC include)
target_compile_features(lu_host PUBLIC cxx_std_17)

add_library(lu_device src/lu_device.cu)
target_include_directories(lu_device PUBLIC include)
set_target_properties(lu_device PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
target_link_libraries(lu_device PRIVATE CUDA::cudart CUDA::cublas)

# Link this target from new CUDA experiments to get the common dependencies.
add_library(factorisation_dependencies INTERFACE)
target_link_libraries(factorisation_dependencies INTERFACE
  lu_reference lu_host lu_device CUDA::cudart CUDA::cublas)

add_executable(lu_example src/lu_example.cpp)
target_link_libraries(lu_example PRIVATE factorisation_dependencies)
target_compile_features(lu_example PRIVATE cxx_std_17)

include(CTest)
if(BUILD_TESTING)
  add_executable(cuda_utilities_test src/cuda_utilities_test.cpp)
  target_include_directories(cuda_utilities_test PRIVATE include)
  target_link_libraries(cuda_utilities_test PRIVATE CUDA::cudart CUDA::cublas)
  target_compile_features(cuda_utilities_test PRIVATE cxx_std_17)
  add_test(NAME cuda_utilities_cpu COMMAND cuda_utilities_test --cpu-only)
  add_test(NAME cuda_check_exit COMMAND cuda_utilities_test --cuda-failure)
  add_test(NAME cublas_check_exit COMMAND cuda_utilities_test --cublas-failure)
  set_tests_properties(cuda_check_exit cublas_check_exit PROPERTIES WILL_FAIL TRUE)
  add_executable(lu_reference_test src/lu_reference_test.cpp)
  target_link_libraries(lu_reference_test PRIVATE lu_reference)
  add_test(NAME lu_cpu_reference COMMAND lu_reference_test)
  add_test(NAME lu_random_cpu COMMAND lu_example --size 17 --cpu-only)
  add_test(NAME lu_random_scalar COMMAND lu_example --size 1 --cpu-only)
  add_executable(lu_factorisation_test src/lu_factorisation_test.cpp)
  target_link_libraries(lu_factorisation_test PRIVATE factorisation_dependencies)
  target_compile_features(lu_factorisation_test PRIVATE cxx_std_17)
  add_test(NAME lu_factorisation_cpu COMMAND lu_factorisation_test --cpu-only)
  option(GPU_FACTORISATIONS_TEST_GPU "Register tests requiring an NVIDIA GPU" OFF)
  if(GPU_FACTORISATIONS_TEST_GPU)
    add_test(NAME lu_factorisation_gpu COMMAND lu_factorisation_test)
    add_test(NAME cuda_utilities_gpu COMMAND cuda_utilities_test)
    add_test(NAME lu_random_gpu COMMAND lu_example --size 17)
  endif()
endif()
